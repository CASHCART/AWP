// File: server.js
const express = require('express');
const fs = require('fs');
const path = require('path');
const bp = require('body-parser');
const jwt = require('jsonwebtoken');
const app = express();
const port = process.env.PORT || 3000;
const dataFile = path.join(__dirname, 'students.json');
if (!fs.existsSync(dataFile)) fs.writeFileSync(dataFile, '[]');
let stu = JSON.parse(fs.readFileSync(dataFile, 'utf8'));
if (!Array.isArray(stu)) throw new Error('students.json must contain an array.');

app.use(bp.urlencoded({ extended: true }));
app.use(bp.json());

app.get('/', (req, res) => {
    res.send('<h3>JWT Protected Express REST API</h3>');
});

const SECRET_KEY = process.env.JWT_SECRET || 'mysecretkey';

app.post('/login', (req, res) => {
    const { username, password } = req.body || {};
    if (username === 'admin' && password === '12345') {
        const token = jwt.sign({ user: username }, SECRET_KEY, { expiresIn: '1h' });
        return res.json({ token });
    }
    res.status(401).send('Invalid credentials');
});

function verifyToken(req, res, next) {
    const bearerHeader = req.headers.authorization;
    if (!bearerHeader) return res.status(403).send('Token missing');
    const parts = bearerHeader.split(' ');
    if (parts.length !== 2 || parts[0] !== 'Bearer' || !parts[1]) {
        return res.status(403).send('Invalid token');
    }
    jwt.verify(parts[1], SECRET_KEY, (err, authData) => {
        if (err) return res.status(403).send('Invalid token');
        req.user = authData;
        next();
    });
}

app.get('/list', verifyToken, (req, res) => {
    res.json(stu);
});

app.post('/', verifyToken, (req, res) => {
    const newstu = req.body;
    if (!newstu || !String(newstu.roll ?? '').trim() || !String(newstu.name ?? '').trim()) {
        return res.status(400).send('Roll and name are required');
    }
    if (stu.some(s => String(s.roll) === String(newstu.roll))) {
        return res.status(409).send('Roll number already exists');
    }
    mywrite([...stu, newstu]);
    res.status(201).send('Inserted');
});

app.put('/', verifyToken, (req, res) => {
    const upstu = req.body;
    if (!upstu || !String(upstu.roll ?? '').trim()) {
        return res.status(400).send('Roll is required');
    }
    const index = stu.findIndex(s => String(s.roll) === String(upstu.roll));
    if (index === -1) return res.status(404).send('Student not found');
    const data = stu.map((s, i) => i === index ? { ...s, ...upstu } : s);
    mywrite(data);
    res.send('Updated successfully');
});

app.delete('/', verifyToken, (req, res) => {
    const delstu = req.body;
    if (!delstu || !String(delstu.roll ?? '').trim()) {
        return res.status(400).send('Roll is required');
    }
    const data = stu.filter(s => String(s.roll) !== String(delstu.roll));
    if (data.length === stu.length) return res.status(404).send('Student not found');
    mywrite(data);
    res.send('Deleted successfully');
});

app.use((req, res) => {
    res.status(404).send('Route not found');
});

app.use((err, req, res, next) => {
    console.error(err);
    res.status(err.status || 500).send('Unable to process request');
});

function mywrite(data) {
    fs.writeFileSync(dataFile, JSON.stringify(data, null, 2));
    stu = data;
}

app.listen(port, () => {
    console.log(`Server running on port ${port}`);
});

Save the code as server.js in this folder. The program creates students.json automatically. Start the server, then open http://localhost:3000; use Ctrl+C to stop it. Send student fields as JSON or URL-encoded form data to /; GET /list displays records. POST /login with {"username":"admin","password":"12345"}; include the returned JWT as Authorization: Bearer TOKEN on CRUD requests.

```bash
node server.js
```
